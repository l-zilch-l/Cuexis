TEST_CASE("R3c real allocation failures preserve fault prefix and discard staged owners",
          "[r3c-fault]") {
    const auto stage = GENERATE(2, 3, 4);
    Fixture3 f;
    expandTickFixture(f, 256);
    auto actual = f.session();
    auto reference = f.session();
    detail::KernelTestControls controls{};
    controls.failAdvanceFinalization = stage == 2;
    if (stage != 2) {
        controls.failedTick = Tick{1000};
        controls.failurePoint = stage == 3 ? detail::KernelFailurePoint::beforeFold
                                          : detail::KernelFailurePoint::beforeSeal;
    }
    REQUIRE(detail::KernelTestAccess::inject(reference, controls));
    const auto h = stage == 2 ? 904 : 1003;
    tick_probe::plan = {};
    tick_probe::plan.failureStage = stage;
    tick_probe::plan.failureTick = stage == 2 ? h : 1000;
    auto result = actual.advance(Tick{h});
    const auto observed = tick_probe::plan;
    tick_probe::plan = {};
    REQUIRE_FALSE(result);
    REQUIRE(observed.triggered);
    REQUIRE(observed.events[1] == 1);
    CHECK(observed.events[7] == 0);
    REQUIRE_FALSE(reference.advance(Tick{h}));
    auto a = detail::KernelTestAccess::captureDTO(actual);
    auto b = detail::KernelTestAccess::captureDTO(reference);
    REQUIRE(a);
    REQUIRE(b);
    CHECK(sameSnapshot(*a, *b));
    auto archive = actual.archive();
    REQUIRE(archive);
    CHECK(detail::ExecutionKernel::tickPointCount(*archive) == 0);
    CHECK(archive->data().records.size() == 1);
    CHECK(a->kernel.state == KernelSessionState::Faulted);
    auto aw = encodeSnapshot(detail::KernelTestAccess::snapshotValue(*a, archive->dependencies()));
    auto bw = encodeSnapshot(detail::KernelTestAccess::snapshotValue(*b, archive->dependencies()));
    REQUIRE(aw);
    REQUIRE(bw);
    CHECK(*aw == *bw);
}

TEST_CASE("R3c optional allocation misses use exact original replay", "[r3c-fault]") {
    const auto stage = GENERATE(0, 5, 10);
    Fixture3 f;
    expandTickFixture(f, 256);
    auto made = detail::ExecutionKernel::prepare(f.configuration(), f.prepared());
    REQUIRE(made);
    auto writer = std::move(*made);
    if (stage == 0) {
        tick_probe::plan = {};
        tick_probe::plan.failureStage = 0;
        tick_probe::plan.failureTick = 901;
    }
    auto advanced = writer->advance(Tick{1203});
    auto capture = tick_probe::plan;
    tick_probe::plan = {};
    REQUIRE(advanced);
    auto archive = writer->archive();
    if (stage == 0) {
        CHECK(capture.triggered);
        CHECK(capture.events[9] == 1);
        CHECK(capture.events[1] == 0);
        CHECK(detail::ExecutionKernel::tickPointCount(archive) == 0);
    } else REQUIRE(detail::ExecutionKernel::tickPointCount(archive) > 0);
    const ReplayArchive external{archive.anchor(),
                                 std::make_shared<const RecoveryInputs>(archive.dependencies())};
    const auto cut = replayCutAt(archive, Tick{1003});
    tick_probe::plan = {};
    if (stage != 0) tick_probe::plan.failureStage = stage;
    auto actual = detail::ExecutionKernel::seekCandidate(archive, Tick{1003}, nullptr, &cut);
    const auto observed = tick_probe::plan;
    tick_probe::plan = {};
    REQUIRE(actual);
    if (stage != 0) {
        CHECK(observed.triggered);
        CHECK(observed.events[8] == 1);
        CHECK(observed.events[6] == 0);
    }
    CHECK((*actual)->tickRestoreCount() == 0);
    auto expected = detail::ExecutionKernel::seekCandidate(external, Tick{1003}, nullptr, &cut);
    REQUIRE(expected);
    auto a = (*actual)->snapshot();
    auto b = (*expected)->snapshot();
    REQUIRE(a);
    REQUIRE(b);
    CHECK(sameSnapshot((*a)->dto, (*b)->dto));
    auto aw = encodeSnapshot(detail::KernelTestAccess::snapshotValue((*a)->dto, archive.dependencies()));
    auto bw = encodeSnapshot(detail::KernelTestAccess::snapshotValue((*b)->dto, archive.dependencies()));
    REQUIRE(aw);
    REQUIRE(bw);
    CHECK(*aw == *bw);
}

TEST_CASE("R3c observed source owners release requested heap", "[r3c-memory]") {
    const auto n = GENERATE(1000U, 40000U);
    Fixture3 f;
    expandTickFixture(f, n);
    const auto config = f.configuration();
    const auto prepared = f.prepared();
    // Warm shared preparation/static paths before either measured owner scope.
    { auto warm = detail::ExecutionKernel::prepare(config, prepared); REQUIRE(warm); }
    struct Usage { std::size_t peak{}, retained{}, released{}, points{}; bool success{}; };
    const auto measure = [&](bool capture) {
        Usage u;
        const auto before = tick_probe::live.load();
        tick_probe::peak.store(before);
        {
            auto result = detail::ExecutionKernel::prepare(config, prepared);
            if (!result) return u;
            auto advanced = capture ? (*result)->advance(Tick{1203})
                                    : (*result)->advanceForReplay(Tick{1203});
            u.success = static_cast<bool>(advanced);
            auto archive = (*result)->archive();
            u.points = detail::ExecutionKernel::tickPointCount(archive);
            result->reset();
            u.retained = tick_probe::live.load() - before;
            u.peak = tick_probe::peak.load() - before;
        }
        u.released = tick_probe::live.load() - before;
        return u;
    };
    const auto baseline = measure(false);
    const auto candidate = measure(true);
    REQUIRE(baseline.success);
    REQUIRE(candidate.success);
    CHECK(baseline.points == 0);
    CHECK(candidate.points > 0);
    CHECK(baseline.released == 0);
    CHECK(candidate.released == 0);
    std::cout << "{\\"event\\":\\"r3c.requested_heap\\",\\"n\\":" << n
              << ",\\"baseline_peak\\":" << baseline.peak
              << ",\\"candidate_peak\\":" << candidate.peak
              << ",\\"baseline_retained\\":" << baseline.retained
              << ",\\"candidate_retained\\":" << candidate.retained
              << ",\\"candidate_points\\":" << candidate.points
              << ",\\"released_delta\\":" << candidate.released << "}\\n";
}
