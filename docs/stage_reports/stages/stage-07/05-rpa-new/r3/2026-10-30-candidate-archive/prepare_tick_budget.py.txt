struct TickPayloadBudget final {
    std::atomic<std::size_t> live{};
    std::size_t limit{128 * 1024 * 1024};
    auto acquire(std::size_t bytes) noexcept -> bool {
        if (bytes > limit) return false;
        auto value = live.load(std::memory_order_relaxed);
        while (value <= limit - bytes) {
            if (live.compare_exchange_weak(value, value + bytes, std::memory_order_relaxed))
                return true;
        }
        return false;
    }
};
