void checkpoint(const char* label) {
#if defined(CUEXIS_ALLOCATION_LIFETIME_OBSERVER)
    mem::checkpoint(label);
#else
    PROCESS_MEMORY_COUNTERS_EX value{};
    if (!GetProcessMemoryInfo(GetCurrentProcess(), reinterpret_cast<PROCESS_MEMORY_COUNTERS*>(&value), sizeof(value)))
        throw std::runtime_error("GetProcessMemoryInfo failed");
    std::printf("R3_PROCESS_MEMORY {\"label\":\"%s\",\"private_commit_bytes\":%llu,\"peak_pagefile_bytes\":%llu,\"working_set_bytes\":%llu,\"peak_working_set_bytes\":%llu}\n",
                label, static_cast<unsigned long long>(value.PrivateUsage),
                static_cast<unsigned long long>(value.PeakPagefileUsage),
                static_cast<unsigned long long>(value.WorkingSetSize),
                static_cast<unsigned long long>(value.PeakWorkingSetSize));
#endif
}
auto hexDigest(const std::vector<std::byte>& bytes) -> std::string {
    const auto hash = core::detail::sha256(bytes);
    std::string result;
    for (const auto b : hash) {
        constexpr char digits[] = "0123456789abcdef";
        const auto value = static_cast<unsigned>(b);
        result.push_back(digits[value >> 4]);result.push_back(digits[value & 15]);
    }
    return result;
}
