from pathlib import Path
import json, shutil, subprocess, hashlib

root = Path.cwd()
base = Path(__file__).resolve().parent
old = root / 'out/verification-20261030/r3-next-v1'
out = base / 'wip-build-v1'
src = base / 'wip-source-v1'
rel = 'engine/presentation_renderer/src/draw_command.cpp'

# 1. Source pair: committed HEAD blob (measured/selected) versus live worktree (refinement).
src.mkdir(parents=True, exist_ok=True)
blob = subprocess.run(['git', 'cat-file', 'blob', 'HEAD:' + rel], capture_output=True, check=True).stdout
(src / 'draw_command_head.cpp').write_bytes(blob)
shutil.copyfile(root / rel, src / 'draw_command_wip.cpp')
shutil.copyfile(old / 'draw_transfer_characterization.cpp', base / 'wip-characterization.cpp')

sha = lambda p: hashlib.sha256(Path(p).read_bytes()).hexdigest()
lf_sha = lambda p: hashlib.sha256(Path(p).read_bytes().replace(b'\r\n', b'\n')).hexdigest()
blob_sha1 = subprocess.run(['git', 'rev-parse', 'HEAD:' + rel], capture_output=True, text=True, check=True).stdout.strip()
head_rehash = subprocess.run(['git', 'hash-object', '--path=' + rel, str(src / 'draw_command_head.cpp')],
                             capture_output=True, text=True, check=True).stdout.strip()
assert head_rehash == blob_sha1, (head_rehash, blob_sha1)
(src / 'binding.json').write_text(json.dumps(dict(
    scope='Baseline is the committed HEAD draw_command.cpp; candidate is the live worktree refinement. No other source differs.',
    baseline_source='git cat-file blob HEAD:' + rel,
    baseline_blob_sha1=blob_sha1,
    baseline_rehash_sha1=head_rehash,
    baseline_sha256=sha(src / 'draw_command_head.cpp'),
    baseline_lf_sha256=lf_sha(src / 'draw_command_head.cpp'),
    baseline_bytes=len(blob),
    candidate_sha256=sha(src / 'draw_command_wip.cpp'),
    candidate_lf_sha256=lf_sha(src / 'draw_command_wip.cpp'),
    candidate_bytes=(src / 'draw_command_wip.cpp').stat().st_size,
    on_disk_selected_crlf_sha256='45dc935f891d2811c90640ec6006c91c5f3d4fe8fd64597859ea83381249cd03',
    on_disk_candidate_crlf_sha256='634157c2f12da2e15c6974d9d09014240070ff976b9bebc181a11848e3807b2e',
    characterization_sha256=sha(base / 'wip-characterization.cpp'),
    identity_proof='git hash-object --path of the extracted blob equals git rev-parse HEAD:path, so the compiled baseline is byte-identical to the committed source.'
), indent=2), encoding='utf-8')

# 2. Reuse the exact product-sequence Release build commands (same common libs/headers as the selected run).
script = (old / 'build_gl_state_v3_generated.py').read_text()
script = script[:script.index('adapter_objects=[]')]
script = script.replace("out=base/'gl-state-build-v3'", "out=Path(r'" + str(out) + "')")
ns = dict(__name__='wip_builder', __file__=str(old / 'build_gl_state_v3_generated.py'))
exec(compile(script, str(old / 'build_gl_state_v3_generated.py'), 'exec'), ns)
compile_tu = ns['compile']
run = ns['run']

baseline_obj = compile_tu('draw_command.cpp', src / 'draw_command_head.cpp', 'draw-baseline')
candidate_obj = compile_tu('draw_command.cpp', src / 'draw_command_wip.cpp', 'draw-candidate')
char_obj = compile_tu('presentation_renderer_tests.cpp', base / 'wip-characterization.cpp', 'neutral-characterization')

neutral_tmpl = next(c['args'] for c in json.loads((old / 'draw-transfer-tests-v2/commands.json').read_text())
                    if c['label'] == 'link-candidate')
opengl_tmpl = next(c['args'] for c in json.loads((old / 'draw-transfer-regression-v1/commands.json').read_text())
                   if c['label'] == 'link-opengl')

def relink(template, exe, obj, label):
    args = []
    swapped = 0
    for argument in template:
        if argument.startswith('/OUT:'):
            args.append('/OUT:' + str(out / exe))
        elif argument.endswith('draw-candidate.obj') or argument.endswith('draw-final.obj'):
            args.append(obj)
            swapped += 1
        elif argument.endswith('neutral-characterization.obj'):
            args.append(char_obj)
        else:
            args.append(argument)
    assert swapped == 1, (label, swapped)
    run(args, label)

for variant, obj in (('baseline', baseline_obj), ('candidate', candidate_obj)):
    relink(neutral_tmpl, 'neutral-' + variant + '.exe', obj, 'link-neutral-' + variant)
    relink(opengl_tmpl, 'opengl-' + variant + '.exe', obj, 'link-opengl-' + variant)

# Physical GPU probe binaries: same recipe as the selected draw-transfer run, but the probe is
# recompiled from the current committed source instead of reusing an archived object.
probe_obj = compile_tu('gpu_throughput_probe.cpp', root / 'tests/player/gpu_throughput_probe.cpp', 'gpu-probe')
gpu_tmpl = next(c['args'] for c in json.loads((old / 'gl-startup-build-v2/commands.json').read_text())
                if c['label'] == 'link-baseline')
parse_obj = str(old / 'json-sax-build-v2/parse-candidate.obj')
for variant, obj in (('baseline', baseline_obj), ('candidate', candidate_obj)):
    args = []
    for argument in gpu_tmpl:
        if argument.startswith('/OUT:'):
            args.append('/OUT:' + str(out / ('gpu-' + variant + '.exe')))
        elif argument.endswith('gpu-probe.obj'):
            args.append(probe_obj)
        else:
            args.append(argument)
    assert any(a == probe_obj for a in args)
    run([args[0], obj, parse_obj, *args[1:]], 'link-gpu-' + variant)

for dll in (old / 'draw-transfer-final-v1').glob('*.dll'):
    shutil.copyfile(dll, out / dll.name)

(out / 'binding.json').write_text(json.dumps(dict(
    scope='Only the neutral draw CPP differs between the two physical binaries; same product-sequence Release libraries, same characterization object, same app-local Release DLLs. Focused regression and equivalence check, not a fresh whole-repo matrix.',
    source_binding_sha256=sha(src / 'binding.json'),
    characterization_sha256=sha(base / 'wip-characterization.cpp'),
    probe_source_sha256=sha(root / 'tests/player/gpu_throughput_probe.cpp'),
    commands_sha256=sha(out / 'commands.json'),
    binaries={p.name: sha(p) for p in out.glob('*.exe')}
), indent=2), encoding='utf-8')

# 3. Characterization: 24 source permutations must print identical order/digest/owning payload.
import os
archive = old / 'draw-transfer-final-v1/test-neutral.log'
archived_lines = [l for l in archive.read_text(encoding='utf-8').splitlines() if l.startswith('DRAW_TRANSFER')]
assert len(archived_lines) == 24
corpus = {}
check = out / 'check'
check.mkdir(exist_ok=True)
for variant in ('baseline', 'candidate'):
    log = check / ('neutral-' + variant + '.log')
    with log.open('x', encoding='utf-8') as stream:
        result = subprocess.run([str(out / ('neutral-' + variant + '.exe')), '--rng-seed', '1'],
                                cwd=root, stdout=stream, stderr=subprocess.STDOUT)
    content = log.read_text(encoding='utf-8', errors='replace')
    assert result.returncode == 0 and 'All tests passed' in content, content[-2000:]
    corpus[variant] = [l for l in content.splitlines() if l.startswith('DRAW_TRANSFER')]
    print(variant, content.strip().splitlines()[-1], flush=True)

assert len(corpus['baseline']) == 24 and len(corpus['candidate']) == 24
assert corpus['baseline'] == corpus['candidate'], 'baseline/candidate characterization diverged'
assert corpus['candidate'] == archived_lines, 'candidate diverges from archived selected evidence'

env = dict(os.environ)
env['SDL_VIDEODRIVER'] = 'dummy'
log = check / 'opengl-candidate.log'
with log.open('x', encoding='utf-8') as stream:
    result = subprocess.run([str(out / 'opengl-candidate.exe'), '--rng-seed', '1'],
                            cwd=root, env=env, stdout=stream, stderr=subprocess.STDOUT)
opengl_text = log.read_text(encoding='utf-8', errors='replace')
assert result.returncode == 0 and 'All tests passed' in opengl_text, opengl_text[-2000:]
print('opengl', opengl_text.strip().splitlines()[-1], flush=True)

assertions = {}
for variant in ('baseline', 'candidate'):
    text = (check / ('neutral-' + variant + '.log')).read_text(encoding='utf-8')
    assertions[variant] = text.strip().splitlines()[-1]
assertions['opengl-candidate'] = opengl_text.strip().splitlines()[-1]

(check / 'decision.json').write_text(json.dumps(dict(
    permutations=24,
    order_digest_and_owning_payload_equal_between_variants=True,
    candidate_matches_archived_selected_evidence=True,
    neutral_assertions=assertions,
    logs={p.name: sha(p) for p in check.glob('*.log')},
    note='Equivalence of observable output and focused regression only. No performance claim is made by this group.'
), indent=2), encoding='utf-8')
print('equivalence ok', flush=True)
