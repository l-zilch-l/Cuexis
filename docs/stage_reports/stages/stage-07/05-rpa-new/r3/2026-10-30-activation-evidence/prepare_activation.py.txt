    expandTickFixture(fixture, count);
    auto writer = detail::ExecutionKernel::prepare(fixture.configuration(), fixture.prepared());

    const auto archive = (*writer)->archive();
    const ReplayArchive external{archive.anchor(),

    writer->reset();
    for (const auto horizon : {852, 853, 854, 902, 904, 1003, 1203}) {
        INFO(count);
        INFO(horizon);
        const auto cut = replayCutAt(archive, Tick{horizon});
        auto actual = detail::ExecutionKernel::seekCandidate(archive, Tick{horizon}, nullptr, &cut);
        auto expected = detail::ExecutionKernel::seekCandidate(external, Tick{horizon}, nullptr, &cut);
        REQUIRE(actual);
        REQUIRE(expected);
        const auto restore = count >= 512 && horizon >= 853 && horizon < 1203;
        if (horizon < 904)
            CHECK((*actual)->tickRestoreCount() == (restore ? 1 : 0));
        auto left = (*actual)->snapshot();
        auto right = (*expected)->snapshot();
        REQUIRE(left);
        REQUIRE(right);
        CHECK(sameSnapshot((*left)->dto, (*right)->dto));
        if (horizon == 853 || horizon == 854) {
            CHECK((*left)->dto.kernel.facts.empty());
            CHECK(std::ranges::any_of((*left)->dto.kernel.phases, [](const auto& phase) {
                return phase.state == KernelPhaseState::Pending;
            }));

        REQUIRE((*actual)->advance(Tick{1203}));
        REQUIRE((*expected)->advance(Tick{1203}));
        left = (*actual)->snapshot();
        right = (*expected)->snapshot();
        REQUIRE(left);
        REQUIRE(right);
        CHECK(sameSnapshot((*left)->dto, (*right)->dto));
        auto actualWire = encodeReplay((*actual)->archive());
        auto expectedWire = encodeReplay((*expected)->archive());
        REQUIRE(actualWire);
        REQUIRE(expectedWire);
        CHECK(*actualWire == *expectedWire);
