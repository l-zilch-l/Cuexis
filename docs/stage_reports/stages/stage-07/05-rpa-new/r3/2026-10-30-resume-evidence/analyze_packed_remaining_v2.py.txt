from pathlib import Path
import json,statistics
out=Path(__file__).resolve().parent/'packed-remaining-measure-v2';rows=json.loads((out/'results.json').read_text())
assert len(rows)==10 and all(r['valid'] for r in rows)
summary=[]
for n,profile in ((1000,'shared'),(10000,'split')):
 group=[r for r in rows if r['n']==n and r['profile']==profile]
 values={}
 for field in ('source','codec'):
  values[field]={}
  for variant in ('baseline','candidate'):
   points=[r['records']['R3_PREPARE'][0] for r in group if r['variant']==variant]
   samples=[p['source_prepare_us'] if field=='source' else p['packed_encode_us']+p['packed_decode_us'] for p in points]
   values[field][variant]=dict(samples=samples,median=statistics.median(samples),relative_range_percent=(max(samples)-min(samples))/statistics.median(samples)*100)
  values[field]['change_percent']=(values[field]['candidate']['median']/values[field]['baseline']['median']-1)*100
 summary.append(dict(n=n,profile=profile,values=values))
retain=summary[1]['values']['codec']['change_percent']<=-10 and summary[0]['values']['codec']['change_percent']<=5 and all(s['values']['source']['change_percent']<=5 for s in summary)
result=dict(retain=retain,summary=summary,scope='Predeclared local candidate gate only; original formal and CI unchanged; byte comparison receipt remains separate')
(out/'decision.json').write_text(json.dumps(result,indent=2));print(json.dumps(result))
