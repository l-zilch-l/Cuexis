from pathlib import Path
import hashlib,json,os,subprocess,time
root=Path.cwd();base=Path(__file__).resolve().parent
out=base/'dependency-canary-v1';out.mkdir(exist_ok=False)
source=out/'source';source.mkdir()
(source/'CMakeLists.txt').write_text('cmake_minimum_required(VERSION 3.25)\nproject(DependencyCanary LANGUAGES CXX)\nadd_executable(canary main.cpp)\n',encoding='ascii')
(source/'value.hpp').write_text('#pragma once\n#define CANARY_VALUE 41\n',encoding='ascii')
(source/'main.cpp').write_text('#include "value.hpp"\nint main(){return CANARY_VALUE;}\n',encoding='ascii')
rows=[]
for lang in ('original','english'):
    build=out/lang;env=os.environ.copy()
    if lang=='english':env['VSLANG']='1033'
    args=['cmake','-S',str(source),'-B',str(build),'-G','Ninja','-DCMAKE_BUILD_TYPE=Release']
    def run(argv,label):
        path=out/(lang+'-'+label+'.log')
        with path.open('xb') as f:r=subprocess.run(argv,env=env,cwd=root,stdout=f,stderr=subprocess.STDOUT)
        if r.returncode:raise RuntimeError((label,r.returncode))
        return path.read_bytes()
    run(args,'configure')
    run(['cmake','--build',str(build),'--parallel','1'],'initial-build')
    cache=dict(l.split('=',1) for l in (build/'CMakeCache.txt').read_text().splitlines() if '=' in l and not l.startswith(('#','//')))
    ninja=cache['CMAKE_MAKE_PROGRAM:FILEPATH']
    deps=subprocess.check_output([ninja,'-C',str(build),'-t','deps'],env=env)
    (out/(lang+'-deps.log')).write_bytes(deps)
    initial=subprocess.run([str(build/'canary.exe')],env=env).returncode
    old=(source/'value.hpp').read_text();value=42 if initial==41 else 41
    (source/'value.hpp').write_text('#pragma once\n#define CANARY_VALUE '+str(value)+'\n',encoding='ascii')
    rebuild=run([ninja,'-C',str(build),'-d','explain'],'header-rebuild')
    actual=subprocess.run([str(build/'canary.exe')],env=env).returncode
    row=dict(language=lang,VSLANG=env.get('VSLANG'),initial=initial,expected_after_header_edit=value,actual_after_header_edit=actual,header_dependency_observed=b'value.hpp' in deps,rebuilt_correctly=actual==value,rules_prefix=[l for l in (build/'CMakeFiles/rules.ninja').read_text(encoding='utf-8').splitlines() if 'msvc_deps_prefix' in l])
    rows.append(row);(out/'results.json').write_text(json.dumps(rows,indent=2),encoding='utf-8');print(json.dumps(row),flush=True)
assert rows[1]['header_dependency_observed'] and rows[1]['rebuilt_correctly']
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
(out/'complete.json').write_text(json.dumps(dict(scope='Actual header edit/rebuild proof; no frozen product build modified. Original locale result retained, English-language environment validated for future new builds.',results_sha256=sha(out/'results.json'),script_sha256=sha(Path(__file__))),indent=2),encoding='utf-8')
