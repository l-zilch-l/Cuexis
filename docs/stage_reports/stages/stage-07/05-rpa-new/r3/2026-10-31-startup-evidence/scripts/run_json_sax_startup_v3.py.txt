from pathlib import Path
import json, shutil, hashlib

base = Path(__file__).resolve().parent
old = base / 'gl-startup-build-v1'
out = base / 'gl-startup-runtime-v2'
out.mkdir(exist_ok=False)
dependencies = Path.cwd() / 'out/verification-20261030/r3-resume-v1/budget-build-v3/vcpkg_installed/x64-windows/bin'
shutil.copyfile(old / 'gpu-baseline.exe', out / 'gpu-baseline.exe')
for path in dependencies.glob('*.dll'): shutil.copyfile(path, out / path.name)
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
(out / 'binding.json').write_text(json.dumps(dict(
    scope='Exact original parser executable with correct app-local Release DLLs; immutable source/binary, corrected runtime dependency binding.',
    parent_binding_sha256=sha(old/'binding.json'),
    files={p.name: sha(p) for p in out.iterdir() if p.suffix in ('.exe','.dll')}
), indent=2), encoding='utf-8')
runner = (base / 'run_gl_startup_response_v2.py').read_text()
start = runner.index('plan = [')
end = runner.index("write(out / 'registration.json'", start)
runner = runner[:start] + "plan = [dict(name='40k-static', n=40000, cancel=False)]\n" + runner[end:]
runner = runner.replace('Finite startup diagnosis, not a throughput group. One 40K static process after small normal/close controls pass.',
    'Finite one-process startup comparison with verified app-local Release DLLs. No ordinary throughput acceptance.')
runner = runner.replace('gl-startup-response-v2', 'json-sax-startup-baseline-v3')
runner = runner.replace("'gl-startup-build-v1/gpu-baseline.exe'", "'gl-startup-runtime-v2/gpu-baseline.exe'")
(base / 'run_json_sax_startup_baseline_v3.py').write_text(runner, encoding='utf-8')
candidate = runner.replace('json-sax-startup-baseline-v3', 'json-sax-startup-candidate-v3')
candidate = candidate.replace("'gl-startup-runtime-v2/gpu-baseline.exe'", "'json-sax-runtime-v3/gpu-sax.exe'")
(base / 'run_json_sax_startup_candidate_v3.py').write_text(candidate, encoding='utf-8')
# Sequential pair; failures stop before proceeding to the next process.
exec(compile(runner, str(base/'run_gl_startup_response_v2.py'), 'exec'))
exec(compile(candidate, str(base/'run_gl_startup_response_v2.py'), 'exec'))
