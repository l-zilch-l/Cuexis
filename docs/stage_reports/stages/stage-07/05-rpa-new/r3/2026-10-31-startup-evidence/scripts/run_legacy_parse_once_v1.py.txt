from pathlib import Path
import shutil, json, hashlib

base=Path(__file__).resolve().parent
out=base/'legacy-parse-once-runtime-v1';out.mkdir(exist_ok=False)
shutil.copyfile(base/'legacy-parse-once-build-v1/prepare-candidate.exe',out/'prepare-candidate.exe')
shutil.copyfile(base/'json-prepare-build-v1/prepare-baseline.exe',out/'prepare-baseline.exe')
for path in (base/'legacy-parse-once-build-v1').glob('*.dll'):shutil.copyfile(path,out/path.name)
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
(out/'binding.json').write_text(json.dumps(dict(scope='Immutable baseline from first SAX-only group; candidate has material code change: SAX plus one parsed legacy tree shared with identity writer. Same probe/libs and exact Release DLLs.',
    candidate_binding_sha256=sha(base/'legacy-parse-once-build-v1/binding.json'),
    baseline_binding_sha256=sha(base/'json-prepare-build-v1/binding.json'),
    files={p.name:sha(p) for p in out.iterdir() if p.suffix in ('.exe','.dll')}),indent=2),encoding='utf-8')
source=(base/'run_json_prepare_v1.py').read_text()
source=source.replace('json-prepare-build-v1','legacy-parse-once-runtime-v1').replace('json-prepare-measure-v1','legacy-parse-once-measure-v1')
source=source.replace('sole change SAX parser','SAX plus legacy parse-once')
exec(compile(source,str(base/'run_json_prepare_v1.py'),'exec'))
