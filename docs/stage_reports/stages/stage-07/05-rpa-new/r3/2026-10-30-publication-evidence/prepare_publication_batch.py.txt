auto PublicationRead::readFacts(std::uint64_t first, std::span<PublicationFact> output) const
    -> core::Result<std::size_t> {
    if (auto valid = validate(); !valid)
        return core::unexpected(std::move(valid.error()));
    if (first < metadata_.fromCursor || first > metadata_.publishableCursor)
        return core::unexpected(publicationError("Fact ordinal outside requested prefix"));
    const auto count = static_cast<std::size_t>(
        std::min<std::uint64_t>(output.size(), metadata_.publishableCursor - first));
    for (std::size_t i = 0; i < count; ++i) {
        const auto& value = projection_->facts[static_cast<std::size_t>(first) + i];
        if (const auto* phase = std::get_if<PhaseOutcomeFact>(&value))
            output[i] = {FactKind::phaseOutcome, phase->factId, &phase->requirement,
                         phase->phase, phase->outcome, phase->error};
        else {
            const auto& receipt = std::get<ReceiptFact>(value);
            output[i] = {receipt.kind, receipt.factId, nullptr, {}, {}, {}};
        }
    }
    return count;
}
'''
